cmake_minimum_required(VERSION 3.22.0)
set(lib_name foxglove)

set(foxglove_SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR})

option(RP26_ENABLE_FOXGLOVE "Build the native Foxglove publisher backend" OFF)

if(RP26_ENABLE_FOXGLOVE)
    file(GLOB foxglove_lib_source CONFIGURE_DEPENDS
        "${foxglove_SOURCE_DIR}/src/*.cpp"
        "${foxglove_SOURCE_DIR}/src/server/*.cpp"
    )
    add_library(${lib_name}_lib SHARED ${foxglove_lib_source})
    target_include_directories(${lib_name}_lib PUBLIC ${foxglove_SOURCE_DIR}/include)
    target_link_libraries(${lib_name}_lib PRIVATE ${foxglove_SOURCE_DIR}/lib/libfoxglove.a)
else()
    add_library(${lib_name}_lib INTERFACE)
    target_include_directories(${lib_name}_lib INTERFACE ${foxglove_SOURCE_DIR}/include)
    message(STATUS "Native Foxglove backend disabled; using no-op visualization publishers")
endif()


add_library(${lib_name}_interface_lib INTERFACE)
target_include_directories(${lib_name}_interface_lib INTERFACE
    ${CMAKE_CURRENT_SOURCE_DIR}/autogenerated_flatbuffers
    ${CMAKE_SOURCE_DIR}/third_party/flatbuffers/include
    ${CMAKE_CURRENT_SOURCE_DIR}/include
)
